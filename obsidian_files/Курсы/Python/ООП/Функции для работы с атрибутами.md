```python
hasattr(object_name, attr_name)#проверяет наличие атрибута у объекта(возвращает булиан)

setattr(object_name, attr_name, attr_value)#уставливает атрибут и его значение объекту

delattr(object_name, attr_name)#удаляет атрибут у объекта

getattr(object_name, attr_name)#получает значение атрибута объекта
```
