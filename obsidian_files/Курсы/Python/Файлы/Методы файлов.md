```python
print(file.name)#возвращает имя файла
print(file.mode)#возвращает режим, в еотором открыт файл
print(file.encoding)#возвращает кодировку файла
print(file.closed)#закрыт ли файл

print(file.readable())#доступен ли для чтения
print(file.writable())#доступен ли для записи
print(file.seekable())#поддерживает ли произвольный доступ(по позиции)
print(file.flush())#обычно файл буферизован, метод записывает весь буфер на диск
file.truncate()#обнуляет файл
f = file.readlines()#помещает в переменную список строк файла
f = file.readline()#считывает строку файла в переменную
```
